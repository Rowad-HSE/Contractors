const SLIDES_TEMPLATE_ID = "1fYjqnHYxirDdxYSHkRAjn9mH5Q1iODPmH2Ns06BGnKc"; const CONTRACTORS_SHEET_NAME = "Form Responses 1"; const VIOLATIONS_SHEET_NAME = "Violations_Log"; const ROOT_FOLDER_NAME = "Rowad_HSE_Violations_PDF"; function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. قراءة المقاولين const sheet = ss.getSheetByName(CONTRACTORS_SHEET_NAME) || ss.getSheets()[0]; const data = sheet.getDataRange().getValues(); const contractors = []; for (let i = 1; i < data.length; i++) { const row = data[i]; if (!row[2] && !row[3]) continue; contractors.push({ id: "GS_" + (i + 1), timestamp: row[0] ? formatDate(row[0]) : "", email: String(row[1] || "").trim(), project: String(row[2] || "").trim(), companyName: String(row[3] || "").trim(), contactName: String(row[4] || "").trim(), contactPhone: String(row[5] || "").trim(), workField: String(row[6] || "").trim(), rowadEngineer: String(row[7] || "").trim() }); } // 2. قراءة سجل المخالفات السابقة وروابط الـ PDF الخاصة بها const violations = []; const logSheet = ss.getSheetByName(VIOLATIONS_SHEET_NAME); if (logSheet) { const vData = logSheet.getDataRange().getValues(); for (let j = vData.length - 1; j >= 1; j--) { const vRow = vData[j]; if (!vRow[2] && !vRow[3]) continue; violations.push({ violationId: String(vRow[0] || ""), project: String(vRow[2] || "").trim(), companyName: String(vRow[3] || "").trim(), violationDate: formatSimpleDate(vRow[4]), violationTime: String(vRow[5] || "").trim(), description: String(vRow[6] || "").trim(), issuedBy: String(vRow[7] || "").trim(), pdfUrl: String(vRow[11] || "").trim() }); } } return createJsonResponse({ status: "success", contractors: contractors, violations: violations }); } catch (err) { return createJsonResponse({ status: "error", message: err.toString() }); } } function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); let logSheet = ss.getSheetByName(VIOLATIONS_SHEET_NAME); if (!logSheet) { logSheet = ss.insertSheet(VIOLATIONS_SHEET_NAME); logSheet.appendRow([ "رقم المخالفة", "تاريخ التسجيل", "المشروع", "اسم الشركة / المقاول", "تاريخ المخالفة", "وقت المخالفة", "وصف المخالفة", "محرر المخالفة", "مسؤول المقاول", "رقم الهاتف", "مهندس الرواد المسؤول", "رابط ملف PDF" ]); logSheet.getRange(1, 1, 1, 12).setFontWeight("bold").setBackground("#1e293b").setFontColor("#ffffff"); } // 1. الوصول للفولدر الرئيسي ثم إنشاء/فتح فولدر فرعي باسم المشروع const rootFolder = getOrCreateFolder(ROOT_FOLDER_NAME, DriveApp); const projectName = (payload.project || "General").trim(); const projectFolder = getOrCreateFolder(projectName, rootFolder); // 2. إنشاء نسخة مؤقتة من القالب داخل فولدر المشروع وتعبئتها const templateFile = DriveApp.getFileById(SLIDES_TEMPLATE_ID); const fileName = `مخالفة_${projectName}_${payload.companyName}_${payload.violationDate}`; const copyFile = templateFile.makeCopy(fileName + "_temp", projectFolder); const presentation = SlidesApp.openById(copyFile.getId()); const slides = presentation.getSlides(); if (slides.length >= 1) { const slide1 = slides[0]; slide1.replaceAllText("{{CONTRACTOR_NAME}}", payload.companyName || ""); slide1.replaceAllText("{{Project_NAME}}", projectName); slide1.replaceAllText("{{VIOLATION_DATE}}", payload.violationDate || ""); slide1.replaceAllText("{{VIOLATION_TIME}}", payload.violationTime || ""); } if (slides.length >= 2) { const slide2 = slides[1]; slide2.replaceAllText("{{VIOLATION_DESCRIPTION}}", payload.description || ""); if (payload.image1Base64) { replacePlaceholderWithImage(slide2, "{{EVIDENCE_IMAGE_1}}", payload.image1Base64); } else { slide2.replaceAllText("{{EVIDENCE_IMAGE_1}}", ""); } if (payload.image2Base64) { replacePlaceholderWithImage(slide2, "{{EVIDENCE_IMAGE_2}}", payload.image2Base64); } else { slide2.replaceAllText("{{EVIDENCE_IMAGE_2}}", ""); } } presentation.saveAndClose(); // 3. حفظ ملف الـ PDF داخل فولدر المشروع وحذف ملف السلايد المؤقت const pdfBlob = copyFile.getAs(MimeType.PDF).setName(fileName + ".pdf"); const pdfFile = projectFolder.createFile(pdfBlob); pdfFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); const pdfUrl = pdfFile.getUrl(); copyFile.setTrashed(true); logSheet.appendRow([ payload.violationId || ("VIO-" + Date.now()), new Date(), projectName, payload.companyName || "", payload.violationDate || "", payload.violationTime || "", payload.description || "", payload.issuedBy || "", payload.contactName || "", payload.contactPhone || "", payload.rowadEngineer || "", pdfUrl ]); return createJsonResponse({ status: "success", pdfUrl: pdfUrl }); } catch (err) { return createJsonResponse({ status: "error", message: err.toString() }); } } // دالة إنشاء أو جلب فولدر داخل فولدر أب (أو Drive الرئيسي) function getOrCreateFolder(folderName, parent) { const folders = parent.getFoldersByName(folderName); if (folders.hasNext()) return folders.next(); return parent.createFolder(folderName); } function replacePlaceholderWithImage(slide, placeholderText, base64DataUrl) { const shapes = slide.getShapes(); const base64Parts = base64DataUrl.split(","); const mimeType = base64Parts[0].match(/:(.*?);/)[1]; const bytes = Utilities.base64Decode(base64Parts[1]); const blob = Utilities.newBlob(bytes, mimeType, "evidence.jpg"); for (let i = 0; i < shapes.length; i++) { const shape = shapes[i]; if (shape.getText && shape.getText().asString().indexOf(placeholderText) !== -1) { const left = shape.getLeft(); const top = shape.getTop(); const width = shape.getWidth(); const height = shape.getHeight(); shape.remove(); slide.insertImage(blob, left, top, width, height); return; } } } function formatDate(val) { if (val instanceof Date) { return Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm"); } return String(val); } function formatSimpleDate(val) { if (val instanceof Date) { return Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd"); } return String(val || "").trim(); } function createJsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
